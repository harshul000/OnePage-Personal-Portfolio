The Report

WITH cte AS (
   SELECT s.name, g.grade, s.marks
   FROM students s
   JOIN grades g
     ON s.marks BETWEEN g.min_mark AND g.max_mark
)
SELECT
   CASE WHEN grade >= 8 THEN name ELSE NULL END AS name,
   grade,
   marks
FROM cte
ORDER BY
   grade DESC,
   CASE WHEN grade >= 8 THEN name end ,
   case when grade  < 8 then marks end

Ollivander's Inventory
select id , age , cte.coins_needed ,cte.power
from (
    select code , power, min(coins_needed) as coins_needed
   from wands
   group by code , power 
)cte join (
   select id , wands.code , power , coins_needed , age , is_evil
   from wands join wands_property on wands.code = wands_property.code
   where is_evil = 0  
)cte1
on cte.code = cte1.code AND
cte.power = cte1.power and cte.coins_needed = cte1.coins_needed
order by power desc ,age desc ;
Approach 2 

SELECT id, age, coins_needed, power
FROM (
   SELECT
       w.id,
       wp.age,
       w.coins_needed,
       w.power,
       ROW_NUMBER() OVER (
           PARTITION BY w.power, wp.age
           ORDER BY w.coins_needed
       ) AS rn
   FROM Wands w
   JOIN Wands_Property wp
       ON w.code = wp.code
   WHERE wp.is_evil = 0
) t
WHERE rn = 1
ORDER BY power DESC, age DESC;

   group by challenges.hacker_id ,name
),

   from cte

Contest Leaderboard
SELECT id, age, coins_needed, power
FROM (
   SELECT
       w.id,
       wp.age,
       w.coins_needed,
       w.power,
       ROW_NUMBER() OVER (
           PARTITION BY w.power, wp.age
           ORDER BY w.coins_needed
       ) AS rn
   FROM Wands w
   JOIN Wands_Property wp
       ON w.code = wp.code
   WHERE wp.is_evil = 0
) t
WHERE rn = 1
ORDER BY power DESC, age DESC;

from Functions f1 join Functions f2
ON f1.x = f2.y and
f2.x = f1.y
group by  f1.x , f1.y

SET NOCOUNT ON;
with cte as (
   select challenge_id ,
   sum(total_submissions) as ts,
   sum(total_accepted_submissions) as tas
   from submission_stats
   group by challenge_id
),
cte1 as (
   select challenge_id ,
   sum(total_views) as tv,
   sum(total_unique_views) as tav
   from view_stats
   group by challenge_id
)
select
cl.contest_id ,
co.hacker_id ,
co.name,
coalesce(sum(ts),0) as ts,
coalesce(sum(tas),0) as tas,
coalesce(sum(tv),0) as tv,
coalesce(sum(tav),0) as tav
from challenges ch join colleges cl ON cl.college_id  = ch.college_id
join contests co on co.contest_id = cl.contest_id
left join cte on cte.challenge_id = ch.challenge_id
left join cte1 on cte1.challenge_id = ch.challenge_id
GROUP by cl.contest_id ,co.hacker_id ,co.name
having coalesce(sum(ts),0) +
coalesce(sum(tas),0) +
coalesce(sum(tv),0) +
coalesce(sum(tav),0) != 0
order by cl.contest_id ;
go


15 Days of Learning SQL

with cte as (
   select distinct submission_date , hacker_id
   from submissions
),
cte1 as (
   select submission_date , hacker_id ,
   count(*)over(partition by hacker_id  order by submission_date) as a
   from cte
),
cte2 as(
   select *,dense_rank() over(order by submission_date) as b
   from cte1
),
cte3 as (
   select submission_date , count(*) as t
   from cte2
   where a = b
   group by submission_date
),
cte4 as (
   select submission_date , submissions.hacker_id ,name ,
   row_number()over(partition by submission_date order by count(*) desc,submissions.hacker_id) as rn
   from submissions join hackers on hackers.hacker_id = submissions.hacker_id
   group by submission_date , submissions.hacker_id ,name
)
select cte4.submission_date ,cte3.t ,cte4.hacker_id , cte4.name
from cte3 join cte4 on cte3.submission_date = cte4.submission_date
where rn=1
order by cte4.submission_date

Occupations

with cte as (
   select name , occupation ,
   row_number() over(partition by occupation order by name) as rn
   from OCCUPATIONS
)SELECT
max(case when occupation = 'Doctor' then name else null end) as Doctor ,
max(case when occupation = 'Professor' then name else null end) as Professor,
max(case when occupation = 'Singer' then name else null end) as Singer,
max(case when occupation = 'Actor' then name else null end) as Actor
from cte
group by rn ;


Binary Tree Nodes
select n,
case when p is null then 'Root'
when n in (select p from bst) then 'Inner' else 'Leaf' end as node
from bst
order by n;
